ชั้น2

อ่านเฉพาะแท็บ ไม่เปิดทั้งไฟล์

ใช้วิธี fetchTabValues_() ที่ NOVA ใช้อยู่แล้ว — ยิง Sheets API ขอแท็บเดียว แทนการ SpreadsheetApp.openById() ซึ่งจะโหลดทั้งไฟล์ ทุกแท็บ ทุกสูตร ทุกฟอร์แมต

// ❌ ช้า — โหลดทั้งไฟล์ 8 แท็บ เพื่อจะอ่านแท็บเดียว
var sh = SpreadsheetApp.openById(fid).getSheetByName('PO_LINE');
var v  = sh.getDataRange().getValues();

// ✅ เร็ว — ขอเฉพาะแท็บที่ต้องใช้
var v = fetchTabValues_(fid, 'PO_LINE');

ไฟล์รายปีมี ~8 แท็บ → วิธีนี้ประหยัดไปประมาณ 7 ใน 8 ของงานที่ไม่จำเป็น

อ่านเฉพาะช่วงแถวที่จะโชว์ (แบ่งหน้า)

หน้าจอโชว์ได้จริงทีละ ~50 แถว → ดึงทีละ 200 แถว/หน้า ไม่ใช่ดึง 6,900 แถวมากองไว้

  • ค้นหา / กรอง → ทำที่ตารางดัชนี (แถวสั้น ๆ 6 คอลัมน์) แล้วค่อยไปหยิบแถวเต็มเฉพาะที่ตรง
  • ห้ามส่งข้อมูลทั้งปีไปให้เบราว์เซอร์ประมวลผลเอง — เครื่องพนักงานจะค้าง

เปิดไฟล์ปีเก่า — เฉพาะเมื่อมีคนกดขอจริง

ระบบไม่รู้จัก File ID ของปีไหนเลยในโค้ด — ถามจาก REGISTRY เอา (เพิ่มปีใหม่ = เพิ่ม 1 แถวใน REGISTRY ไม่ต้องแก้โค้ดเลยตลอด 20 ปี)

function yearFile_(yearTH){                 // 2569 → File ID
  var reg = cacheOr_('REGISTRY_PS', 21600, function(){
    return fetchTabValues_(CFG.MASTER,'REGISTRY');
  });
  for (var i=1;i<reg.length;i++){
    if (String(reg[i][0]).trim()===String(yearTH)
     && String(reg[i][3]).toUpperCase()==='PS'
     && String(reg[i][6]).toUpperCase()!=='N') return reg[i][4];
  }
  throw new Error('ยังไม่ได้ลงทะเบียนไฟล์ของปี '+yearTH+' ใน REGISTRY');
}

วิธีทำให้ “กดบันทึก” เร็ว

  • เขียนก่อน ตอบทันที — บันทึก GRN แล้วตอบกลับเลย ส่วนงานต่อเนื่อง (อัปเดต OPEN_PO, ยอดรวม, แจ้งเตือน LINE) ให้ทำเบื้องหลัง
  • เขียนเป็นก้อนเดียว — setValues() ครั้งเดียวทั้งบล็อก ห้ามวน setValue() ทีละเซลล์ (ช้ากว่าหลักร้อยเท่า)
  • อัปโหลดรูปแยกจากการบันทึก — กดบันทึกเสร็จก่อน รูปค่อยทยอยขึ้น พร้อมย่อรูปเหลือ ~300 KB ที่เครื่องก่อนส่ง
  • Append-only — เพิ่มแถวท้ายตารางเสมอ ไม่ค้นหาแถวเดิมมาแก้ (แก้ = ออกแถวกลับรายการพร้อมเหตุผล)
